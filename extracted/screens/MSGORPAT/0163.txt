{ -1 >R BEGIN DUP -1 = IF DROP 1 ELSE >R 0 THEN END             
  BEGIN R> DUP -1 = IF DROP 1 ELSE } B, { 0 THEN END BASE ! ; } 
{ : QUADPAT } 4 PBASE ! { ; } { : BINPAT } 2 PBASE ! { ; }      
{ : ~ } { BASE @ -1 } PBASE { @ BASE ! ; } ( START OF PATTERN ) 
{ : PATTERN } DECIMAL DATA { ; } { : QUAD } 4 BASE ! { ; }
